SERVER-SENT EVENTS LIVE LOG VIEWER
Setup steps from scratch on macOS
==================================

WHAT YOU ARE BUILDING
---------------------
- A small SSE (Server-Sent Events) server in Node.js on port 4000.
- A React + TypeScript frontend on port 3000.
- The frontend connects to the SSE stream, displays live logs, filters by
  severity (ALL / INFO / WARN / ERROR), and reconnects with exponential backoff.
- Vitest + React Testing Library tests.

STEP 7 - CREATE src/LiveLogViewer.tsx
-------------------------------------
File: src/LiveLogViewer.tsx

import React, { useState, useEffect, useRef } from 'react';

// TypeScript Interface for Log Entries
export interface LogEntry {
    id: string;
    timestamp: string;
    level: 'INFO' | 'WARN' | 'ERROR';
    message: string;
}

type ConnectionStatus = 'CONNECTED' | 'RECONNECTING' | 'DISCONNECTED';

export const LiveLogViewer: React.FC = () => {
    const [logs, setLogs] = useState<LogEntry[]>([]);
    const [filter, setFilter] = useState<'ALL' | 'INFO' | 'WARN' | 'ERROR'>('ALL');
    const [status, setStatus] = useState<ConnectionStatus>('DISCONNECTED');

    const retryDelayRef = useRef<number>(1000); // Initial 1s retry delay
    const eventSourceRef = useRef<EventSource | null>(null);

    useEffect(() => {
        let timerId: ReturnType<typeof setTimeout>;

        const connectSSE = () => {
            setStatus('RECONNECTING');

            // Connect to the mock SSE endpoint
            const es = new EventSource('http://localhost:4000/api/logs');
            eventSourceRef.current = es;

            es.onopen = () => {
                setStatus('CONNECTED');
                retryDelayRef.current = 1000; // Reset exponential backoff on successful connection
            };

            es.onmessage = (event) => {
                try {
                    const newLog: LogEntry = JSON.parse(event.data);
                    setLogs((prevLogs) => [newLog, ...prevLogs.slice(0, 99)]); // Keep latest 100 logs
                } catch (err) {
                    console.error('Failed to parse log event:', err);
                }
            };

            es.onerror = () => {
                setStatus('DISCONNECTED');
                es.close();

                // Exponential backoff logic: 1s -> 2s -> 4s (capped at 8s)
                const nextDelay = retryDelayRef.current;
                retryDelayRef.current = Math.min(retryDelayRef.current * 2, 8000);

                console.log(`Connection lost. Retrying in ${nextDelay / 1000} seconds...`);
                timerId = setTimeout(connectSSE, nextDelay);
            };
        };

        connectSSE();

        // Cleanup on component unmount
        return () => {
            if (eventSourceRef.current) {
                eventSourceRef.current.close();
            }
            clearTimeout(timerId);
        };
    }, []);

    // Filter logs by selected severity level
    const filteredLogs = logs.filter((log) => {
        if (filter === 'ALL') return true;
        return log.level === filter;
    });

    return (
        <div style={{ padding: '20px', fontFamily: 'monospace', maxWidth: '800px', margin: '0 auto' }}>
            <h2>Live System Logs (SSE)</h2>

            {/* Connection Status Badge */}
            <div style={{ marginBottom: '15px' }}>
                <strong>Status: </strong>
                <span
                    style={{
                        padding: '4px 8px',
                        borderRadius: '4px',
                        color: '#fff',
                        backgroundColor:
                            status === 'CONNECTED' ? '#2e7d32' : status === 'RECONNECTING' ? '#ed6c02' : '#d32f2f',
                    }}
                >
                    {status}
                </span>
            </div>

            {/* Severity Filter Buttons */}
            <div style={{ marginBottom: '15px', display: 'flex', gap: '8px' }}>
                {(['ALL', 'INFO', 'WARN', 'ERROR'] as const).map((level) => (
                    <button
                        key={level}
                        onClick={() => setFilter(level)}
                        style={{
                            padding: '6px 12px',
                            cursor: 'pointer',
                            fontWeight: filter === level ? 'bold' : 'normal',
                            border: filter === level ? '2px solid #1976d2' : '1px solid #ccc',
                        }}
                    >
                        {level}
                    </button>
                ))}
            </div>

            {/* Log Feed */}
            <div
                style={{
                    border: '1px solid #ccc',
                    borderRadius: '4px',
                    height: '400px',
                    overflowY: 'auto',
                    padding: '10px',
                    backgroundColor: '#1e1e1e',
                    color: '#f1f1f1',
                }}
            >
                {filteredLogs.length === 0 ? (
                    <p>No log events received yet...</p>
                ) : (
                    filteredLogs.map((log) => (
                        <div key={log.id} style={{ marginBottom: '6px', fontSize: '13px' }}>
                            <span style={{ color: '#888', marginRight: '8px' }}>[{log.timestamp}]</span>
                            <span
                                style={{
                                    fontWeight: 'bold',
                                    marginRight: '8px',
                                    color:
                                        log.level === 'ERROR' ? '#ff5252' : log.level === 'WARN' ? '#ffd740' : '#69f0ae',
                                }}
                            >
                                [{log.level}]
                            </span>
                            <span>{log.message}</span>
                        </div>
                    ))
                )}
            </div>
        </div>
    );
};

STEP 8 - CREATE server.js
-------------------------
File: server.js

const http = require('http');

const PORT = 4000;

const server = http.createServer((req, res) => {
  // Set headers for Server-Sent Events (SSE)
  if (req.url === '/api/logs') {
    res.writeHead(200, {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
      'Connection': 'keep-alive',
      'Access-Control-Allow-Origin': '*', // Allow cross-origin requests from your frontend
    });

    // Send an initial connection message
    const initialLog = {
      id: Date.now().toString(),
      timestamp: new Date().toLocaleTimeString(),
      level: 'INFO',
      message: 'SSE Stream connected successfully.',
    };
    res.write(`data: ${JSON.stringify(initialLog)}\n\n`);

    // Periodically send mock log events every 3 seconds
    const intervalId = setInterval(() => {
      const levels = ['INFO', 'WARN', 'ERROR'];
      const messages = [
        'User authentication successful',
        'Database query latency exceeded threshold',
        'Failed to process transaction payload',
        'Cache invalidated for user session',
        'Incoming webhook received from gateway',
      ];

      const randomLevel = levels[Math.floor(Math.random() * levels.length)];
      const randomMessage = messages[Math.floor(Math.random() * messages.length)];

      const logEntry = {
        id: Date.now().toString(),
        timestamp: new Date().toLocaleTimeString(),
        level: randomLevel,
        message: randomMessage,
      };

      res.write(`data: ${JSON.stringify(logEntry)}\n\n`);
    }, 3000);

    // Clean up interval if client disconnects
    req.on('close', () => {
      clearInterval(intervalId);
      res.end();
    });
  } else {
    res.writeHead(404, { 'Content-Type': 'text/plain' });
    res.end('Not Found');
  }
});

server.listen(PORT, () => {
  console.log(`Mock SSE Server running at http://localhost:${PORT}/api/logs`);
});

STEP 11 - CREATE vitest.setup.ts
--------------------------------
jsdom does not provide EventSource, so the test setup replaces it with a
minimal mock. This file also cleans up the DOM after every test.

File: vitest.setup.ts

import '@testing-library/jest-dom/vitest';
import { afterEach } from 'vitest';
import { cleanup } from '@testing-library/react';

class MockEventSource {
  readonly url: string;
  readonly withCredentials = false;
  readonly CONNECTING = 0;
  readonly OPEN = 1;
  readonly CLOSED = 2;
  readyState = 0;
  onopen: ((this: EventSource, ev: Event) => unknown) | null = null;
  onmessage: ((this: EventSource, ev: MessageEvent) => unknown) | null = null;
  onerror: ((this: EventSource, ev: Event) => unknown) | null = null;

  constructor(url: string | URL) {
    this.url = url.toString();
  }

  close() {
    this.readyState = this.CLOSED;
  }

  addEventListener() {}
  removeEventListener() {}
  dispatchEvent() {
    return true;
  }
}

globalThis.EventSource = MockEventSource as unknown as typeof EventSource;

afterEach(() => {
  cleanup();
});

TROUBLESHOOTING
---------------
1. "Could not find a required file. Name: index.html"
   You are missing public/index.html. Run react-scripts from the project
   root and make sure the file from Step 4 exists.

2. "EADDRINUSE: address already in use :::4000"
   Another process is using port 4000. Find and stop it:

     lsof -ti:4000 | xargs kill

   Then run "npm run start:mock-sse" again.

3. Status stays DISCONNECTED or RECONNECTING
   - Make sure the SSE server is running on port 4000.
   - Check http://localhost:4000/api/logs in a browser.
   - The frontend connects to http://localhost:4000/api/logs exactly.

4. Errors from node_modules/@types/node/ffi.d.ts after running tsc
   Your TypeScript is too old. Install TypeScript 5.9 or newer:

     npm install -D typescript@^5.9.3

5. Tests fail with "EventSource is not defined"
   Make sure vitest.setup.ts exists and is listed in vitest.config.mts
   under test.setupFiles.

6. npm warns about typescript peer dependency with react-scripts@5
   This is expected because react-scripts declares TypeScript ^4. The
   project uses TypeScript 5.9 and the build/test commands work.

7. npm warns about 34 vulnerabilities
   These come from react-scripts@5 and its old dependency tree. Do not run
   "npm audit fix --force" because it can break the toolchain.
